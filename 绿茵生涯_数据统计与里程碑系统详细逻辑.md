## 五、纪录打破检测

```javascript
function checkRecordBreak(player, matchStats) {
  const events = [];
  
  // ===== 俱乐部纪录 =====
  const clubRecords = player.currentClub.records;
  const clubStats = player.stats.club.byClub[player.currentClub.id];
  
  // 俱乐部队史射手王
  if (clubStats.goals > clubRecords.allTime.topScorer.value && !clubRecords.broken.topScorer) {
    events.push({
      type: "record_break",
      record: "俱乐部队史射手王",
      oldHolder: clubRecords.allTime.topScorer.name,
      oldValue: clubRecords.allTime.topScorer.value,
      newValue: clubStats.goals
    });
    clubRecords.broken.topScorer = true;
  }
  
  // 俱乐部队史出场王
  if (clubStats.appearances > clubRecords.allTime.appearances.value && !clubRecords.broken.appearances) {
    events.push({
      type: "record_break",
      record: "俱乐部队史出场王",
      ...
    });
  }
  
  // 单赛季进球纪录
  if (player.stats.currentSeason.goals > clubRecords.allTime.singleSeasonGoals.value) {
    events.push({
      type: "record_break",
      record: "俱乐部单赛季进球纪录",
      ...
    });
  }
  
  // ===== 联赛纪录 =====
  const leagueRecords = player.currentLeague.records;
  if (player.stats.currentSeason.goals > leagueRecords.allTime.singleSeasonGoals.value) {
    events.push({
      type: "record_break",
      record: `${player.currentLeague}单赛季进球纪录`,
      oldHolder: leagueRecords.allTime.singleSeasonGoals.name,
      oldValue: leagueRecords.allTime.singleSeasonGoals.value,
      newValue: player.stats.currentSeason.goals
    });
  }
  
  // ===== 欧战纪录 =====
  if (player.stats.club.byCompetition.championsLeague.goals > europeRecords.championsLeague.allTimeTopScorer.value) {
    events.push({
      type: "record_break",
      record: "欧冠历史射手王",
      oldHolder: "C罗",
      oldValue: 140,
      newValue: player.stats.club.byCompetition.championsLeague.goals
    });
  }
  
  // ===== 国家队纪录 =====
  if (player.stats.nationalTeam.total.goals > nationalRecords.allTime.topScorer.value) {
    events.push({
      type: "record_break",
      record: `${player.country}国家队历史射手王`,
      ...
    });
  }
  
  return events;
}
```

### 7.1 检测引擎

```javascript
function checkMilestones(player, matchStats) {
  const triggeredMilestones = [];
  
  // 进球里程碑
  milestones.goals.forEach(milestone => {
    const currentTotal = player.careerStats.club.total.goals + player.careerStats.nationalTeam.total.goals;
    if (currentTotal >= milestone.value && !player.milestonesReached.includes(milestone.name)) {
      triggeredMilestones.push(milestone);
      player.milestonesReached.push(milestone.name);
    }
  });
  
  // 助攻里程碑
  milestones.assists.forEach(milestone => {
    const currentTotal = player.careerStats.club.total.assists + player.careerStats.nationalTeam.total.assists;
    if (currentTotal >= milestone.value && !player.milestonesReached.includes(milestone.name)) {
      triggeredMilestones.push(milestone);
      player.milestonesReached.push(milestone.name);
    }
  });
  
  // 出场里程碑
  milestones.appearances.forEach(milestone => {
    const currentTotal = player.careerStats.club.total.appearances + player.careerStats.nationalTeam.total.appearances;
    if (currentTotal >= milestone.value && !player.milestonesReached.includes(milestone.name)) {
      triggeredMilestones.push(milestone);
      player.milestonesReached.push(milestone.name);
    }
  });
  
  // 俱乐部里程碑
  milestones.clubGoals.forEach(milestone => {
    if (player.currentClubStats.goals >= milestone.value && !player.milestonesReached.includes(`club_${milestone.name}`)) {
      triggeredMilestones.push(milestone);
      player.milestonesReached.push(`club_${milestone.name}`);
    }
  });
  
  // 国家队里程碑
  milestones.nationalGoals.forEach(milestone => {
    if (player.careerStats.nationalTeam.total.goals >= milestone.value && !player.milestonesReached.includes(`national_${milestone.name}`)) {
      triggeredMilestones.push(milestone);
      player.milestonesReached.push(`national_${milestone.name}`);
    }
  });
  
  // 特殊里程碑
  milestones.special.forEach(milestone => {
    if (checkSpecialCondition(player, milestone.condition) && !player.milestonesReached.includes(milestone.name)) {
      triggeredMilestones.push(milestone);
      player.milestonesReached.push(milestone.name);
    }
  });
  
  return triggeredMilestones;
}
```

```javascript
function triggerMilestoneEvent(player, milestone) {
  const importance = milestone.importance;
  
  // 根据重要性决定事件规模
  const eventScale = {
    low: { mediaCoverage: "local", fanReaction: "small", reputationGain: 1 },
    medium: { mediaCoverage: "national", fanReaction: "medium", reputationGain: 3 },
    high: { mediaCoverage: "national+international", fanReaction: "large", reputationGain: 5 },
    legendary: { mediaCoverage: "global", fanReaction: "huge", reputationGain: 10 }
  };
  
  const scale = eventScale[importance];
  
  // 1. 比赛中/赛后通知
  showMilestoneNotification(player, milestone);
  
  // 2. 生成新闻
  generateMilestoneNews(player, milestone, scale);
  
  // 3. 球迷反应
  triggerFanMilestoneReaction(player, milestone, scale);
  
  // 4. 声望/关系变化
  player.reputation += scale.reputationGain;
  player.fanApproval += scale.reputationGain;
  
  // 5. 重要里程碑触发特殊事件
  if (importance === "legendary") {
    triggerLegendaryMilestoneEvent(player, milestone);
  }
}
```
